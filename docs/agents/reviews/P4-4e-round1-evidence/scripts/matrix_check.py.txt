#!/usr/bin/env python3
import subprocess
import re
import sys

# Get all test names across repository
res = subprocess.run(["go", "test", "-list", ".*", "./..."], capture_output=True, text=True)
all_tests = set()
for line in res.stdout.splitlines():
    line = line.strip()
    if line.startswith("Test"):
        all_tests.add(line)

print(f"Total tests found in repo: {len(all_tests)}")

# Read test-security-run1.txt to see PASS / SKIP / FAIL
sec_results = {}
with open("docs/agents/reviews/P4-4e-round1-evidence/test-security-run1.txt") as f:
    for line in f:
        m = re.match(r"^--- (PASS|SKIP|FAIL): (Test\w+)", line)
        if m:
            status, tname = m.groups()
            sec_results[tname] = status

print(f"Total tests in test-security-run1: {len(sec_results)}")
