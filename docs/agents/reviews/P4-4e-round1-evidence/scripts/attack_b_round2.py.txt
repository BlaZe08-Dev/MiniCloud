#!/usr/bin/env python3
import json
import os
import subprocess
import time
import urllib.request
import urllib.error

# Read test database URL from .env
db_url = ""
with open(".env") as f:
    for line in f:
        if line.startswith("MINICLOUD_TEST_DATABASE_URL="):
            db_url = line.strip().split("=", 1)[1]
            break

port = 18095
addr = f"127.0.0.1:{port}"

# Start coordinator on dedicated port
env = os.environ.copy()
env["MINICLOUD_STORAGE_DRIVER"] = "postgres"
env["MINICLOUD_DATABASE_URL"] = db_url
env["MINICLOUD_HTTP_ADDR"] = addr

proc = subprocess.Popen(["bin/coordinator"], env=env, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
time.sleep(1.5)

def do_req(method, path, body_dict=None, raw_body=None):
    url = f"http://{addr}{path}"
    headers = {"Content-Type": "application/json"}
    if body_dict is not None:
        data = json.dumps(body_dict).encode("utf-8")
    elif raw_body is not None:
        data = raw_body
    else:
        data = None
    req = urllib.request.Request(url, data=data, headers=headers, method=method)
    try:
        with urllib.request.urlopen(req) as resp:
            return resp.status, resp.read().decode("utf-8", errors="replace")
    except urllib.error.HTTPError as e:
        return e.code, e.read().decode("utf-8", errors="replace")

try:
    print("=== 1. Register worker text fields ===")
    status, body = do_req("POST", "/v1/workers/register", {"id":"w_nul_ver","version":"v\u0000","total_slots":1,"supported_tasks":["echo"]})
    print("1a. NUL in version:", status, body.strip())

    status, body = do_req("POST", "/v1/workers/register", {"id":"w_nul_task","version":"v1","total_slots":1,"supported_tasks":["echo\u0000"]})
    print("1b. NUL in supported_tasks:", status, body.strip())

    status, body = do_req("POST", "/v1/workers/register", {"id":"w_long_ver","version":"v"*100,"total_slots":1,"supported_tasks":["echo"]})
    print("1c. Very long version (>64 chars):", status, body.strip())

    # Raw invalid UTF-8 bytes \xff\xfe in version
    raw_bad_utf8 = b'{"id":"w_bad_utf8","version":"\xff\xfe","total_slots":1,"supported_tasks":["echo"]}'
    status, body = do_req("POST", "/v1/workers/register", raw_body=raw_bad_utf8)
    print("1d. Invalid UTF-8 in version registration:", status, body.strip())

    # Read back stored version through API
    status, body = do_req("GET", "/v1/workers")
    print("1e. Read back workers list status:", status)
    workers = json.loads(body).get("workers", [])
    bad_worker = next((w for w in workers if w.get("id") == "w_bad_utf8"), None)
    if bad_worker:
        ver = bad_worker.get("version")
        print(f"    Found w_bad_utf8: version={ver!r}, hex={ver.encode('utf-8').hex()}, unicode_escaped={ver.encode('unicode_escape').decode()}")
    else:
        print("    w_bad_utf8 not found in workers list! Full list:", body.strip())

    print("\n=== 2. Heartbeat worker text fields ===")
    status, body = do_req("POST", "/v1/workers/w%00bad/heartbeat", {"available_slots": 1})
    print("2a. NUL in worker_id path parameter:", status, body.strip())

    print("\n=== 3. Setup fresh worker for result and renew tests ===")
    w_id = "w_redo_b"
    status, body = do_req("POST", "/v1/workers/register", {"id":w_id,"version":"v1","total_slots":20,"supported_tasks":["echo"]})
    print("Register worker w_redo_b:", status)

    def get_fresh_poll():
        # Submit fresh unique job and poll it
        s_stat, s_body = do_req("POST", "/v1/jobs", {"task_type":"echo","payload":{"ts": time.time_ns()}})
        p_stat, p_body = do_req("POST", f"/v1/workers/{w_id}/poll")
        if p_stat != 200:
            raise RuntimeError(f"Poll failed: {p_stat} {p_body}")
        asg = json.loads(p_body, strict=False)["assignment"]
        j_id = asg["job_id"]
        a_id = asg["attempt_id"]
        tok = asg["lease_token"]
        return j_id, a_id, tok

    print("\n=== 4. Report Result text fields (matching job, attempt, token) ===")
    # 4a. NUL in output
    j, a, tok = get_fresh_poll()
    status, body = do_req("POST", f"/v1/jobs/{j}/attempts/{a}/result",
                          {"worker_id":w_id,"lease_token":tok,"exit_code":0,"output":"hello\u0000world"})
    print("4a. NUL in output:", status, body.strip())

    # 4b. NUL in error
    j, a, tok = get_fresh_poll()
    status, body = do_req("POST", f"/v1/jobs/{j}/attempts/{a}/result",
                          {"worker_id":w_id,"lease_token":tok,"exit_code":1,"output":"","error":"fail\u0000boom","error_classification":"PERMANENT"})
    print("4b. NUL in error:", status, body.strip())

    # 4c. NUL in error_classification
    j, a, tok = get_fresh_poll()
    status, body = do_req("POST", f"/v1/jobs/{j}/attempts/{a}/result",
                          {"worker_id":w_id,"lease_token":tok,"exit_code":1,"output":"","error":"boom","error_classification":"PERM\u0000ANENT"})
    print("4c. NUL in error_classification:", status, body.strip())

    # 4d. Output exceeding 64 KiB
    j, a, tok = get_fresh_poll()
    status, body = do_req("POST", f"/v1/jobs/{j}/attempts/{a}/result",
                          {"worker_id":w_id,"lease_token":tok,"exit_code":0,"output":"x"*70000})
    print("4d. Output exceeding 64 KiB (>65536):", status, body.strip())

    # 4e. Error exceeding 10 KiB
    j, a, tok = get_fresh_poll()
    status, body = do_req("POST", f"/v1/jobs/{j}/attempts/{a}/result",
                          {"worker_id":w_id,"lease_token":tok,"exit_code":1,"output":"","error":"x"*15000,"error_classification":"PERMANENT"})
    print("4e. Error exceeding 10 KiB (>10240):", status, body.strip())

    # 4f. Invalid UTF-8 in output
    j, a, tok = get_fresh_poll()
    raw_out = b'{"worker_id":"' + w_id.encode() + b'","lease_token":"' + tok.encode() + b'","exit_code":0,"output":"bad \xff\xfe utf8"}'
    status, body = do_req("POST", f"/v1/jobs/{j}/attempts/{a}/result", raw_body=raw_out)
    print("4f. Invalid UTF-8 in output:", status, body.strip())

    print("\n=== 5. Renew Lease text fields (matching job, attempt) ===")
    # 5a. NUL in lease_token
    j, a, tok = get_fresh_poll()
    status, body = do_req("POST", f"/v1/jobs/{j}/attempts/{a}/renew",
                          {"worker_id":w_id,"lease_token":"tok\u0000bad"})
    print("5a. NUL in lease_token:", status, body.strip())

    # 5b. Very long lease_token
    j, a, tok = get_fresh_poll()
    status, body = do_req("POST", f"/v1/jobs/{j}/attempts/{a}/renew",
                          {"worker_id":w_id,"lease_token":"t"*2000})
    print("5b. Very long lease_token (2000 chars):", status, body.strip())

finally:
    proc.terminate()
    try:
        proc.wait(timeout=5)
    except subprocess.TimeoutExpired:
        proc.kill()
        proc.wait()
