package main

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"os"
	"os/exec"
	"strings"
	"sync"
	"sync/atomic"
	"time"
)

type ResultResp struct {
	Status   string `json:"status"`
	JobState string `json:"job_state"`
}

func main() {
	// Read database URL from .env
	envBytes, err := os.ReadFile(".env")
	if err != nil {
		fmt.Fprintf(os.Stderr, "failed to read .env: %v\n", err)
		os.Exit(1)
	}
	var dbURL string
	for _, line := range strings.Split(string(envBytes), "\n") {
		if strings.HasPrefix(line, "MINICLOUD_TEST_DATABASE_URL=") {
			dbURL = strings.TrimPrefix(line, "MINICLOUD_TEST_DATABASE_URL=")
			break
		}
	}
	if dbURL == "" {
		fmt.Fprintln(os.Stderr, "MINICLOUD_TEST_DATABASE_URL not found")
		os.Exit(1)
	}

	port := 18085
	addr := fmt.Sprintf("127.0.0.1:%d", port)

	// Start coordinator with short lease duration (2s) and fast sweeper (500ms)
	cmd := exec.Command("bin/coordinator")
	cmd.Env = append(os.Environ(),
		"MINICLOUD_STORAGE_DRIVER=postgres",
		"MINICLOUD_DATABASE_URL="+dbURL,
		"MINICLOUD_HTTP_ADDR="+addr,
		"MINICLOUD_LEASE_DURATION=2s",
		"MINICLOUD_SWEEPER_INTERVAL=500ms",
		"MINICLOUD_STARTUP_GRACE_PERIOD=0s",
	)
	cmd.Stdout = io.Discard
	cmd.Stderr = io.Discard
	if err := cmd.Start(); err != nil {
		fmt.Fprintf(os.Stderr, "failed to start coordinator: %v\n", err)
		os.Exit(1)
	}
	defer func() {
		_ = cmd.Process.Kill()
		_ = cmd.Wait()
	}()

	client := &http.Client{Timeout: 10 * time.Second}

	// Wait for ready
	for i := 0; i < 30; i++ {
		resp, err := client.Get(fmt.Sprintf("http://%s/readyz", addr))
		if err == nil && resp.StatusCode == http.StatusOK {
			resp.Body.Close()
			break
		}
		if resp != nil {
			resp.Body.Close()
		}
		time.Sleep(200 * time.Millisecond)
	}

	fmt.Printf("Coordinator running on %s\n", addr)

	// Register 10 workers
	numWorkers := 10
	for i := 0; i < numWorkers; i++ {
		wid := fmt.Sprintf("w_race_%d", i)
		regBody, _ := json.Marshal(map[string]any{
			"id":              wid,
			"version":         "v1",
			"total_slots":     10,
			"supported_tasks": []string{"echo"},
		})
		resp, err := client.Post(fmt.Sprintf("http://%s/v1/workers/register", addr), "application/json", bytes.NewReader(regBody))
		if err != nil || resp.StatusCode != http.StatusOK {
			fmt.Fprintf(os.Stderr, "failed to register %s: %v\n", wid, err)
			os.Exit(1)
		}
		resp.Body.Close()
	}

	fmt.Printf("Registered %d workers\n", numWorkers)

	numJobs := 200
	jobIDs := make([]string, numJobs)
	var submitWg sync.WaitGroup
	submitWg.Add(numJobs)
	for i := 0; i < numJobs; i++ {
		go func(idx int) {
			defer submitWg.Done()
			subBody, _ := json.Marshal(map[string]any{
				"task_type":    "echo",
				"payload":      map[string]string{"msg": fmt.Sprintf("race_%d", idx)},
				"max_attempts": 2,
			})
			resp, err := client.Post(fmt.Sprintf("http://%s/v1/jobs", addr), "application/json", bytes.NewReader(subBody))
			if err != nil || resp.StatusCode != http.StatusCreated {
				fmt.Fprintf(os.Stderr, "submit job %d failed: %v\n", idx, err)
				return
			}
			var j struct {
				ID string `json:"id"`
			}
			_ = json.NewDecoder(resp.Body).Decode(&j)
			resp.Body.Close()
			jobIDs[idx] = j.ID
		}(i)
	}
	submitWg.Wait()
	fmt.Printf("Submitted %d jobs\n", numJobs)

	var (
		commitsSuccess atomic.Int64
		commitsLost    atomic.Int64
		cancelsSuccess atomic.Int64
		cancelsFailed  atomic.Int64
	)

	ctx, cancel := context.WithTimeout(context.Background(), 25*time.Second)
	defer cancel()

	var workerWg sync.WaitGroup
	// Start polling and result workers
	for i := 0; i < numWorkers; i++ {
		wid := fmt.Sprintf("w_race_%d", i)
		workerWg.Add(1)
		go func(workerID string) {
			defer workerWg.Done()
			for {
				select {
				case <-ctx.Done():
					return
				default:
				}
				resp, err := client.Post(fmt.Sprintf("http://%s/v1/workers/%s/poll", addr, workerID), "application/json", nil)
				if err != nil {
					time.Sleep(50 * time.Millisecond)
					continue
				}
				if resp.StatusCode != http.StatusOK {
					resp.Body.Close()
					time.Sleep(50 * time.Millisecond)
					continue
				}
				var p struct {
					Assignment struct {
						JobID      string `json:"job_id"`
						AttemptID  string `json:"attempt_id"`
						LeaseToken string `json:"lease_token"`
					} `json:"assignment"`
				}
				_ = json.NewDecoder(resp.Body).Decode(&p)
				resp.Body.Close()

				asg := p.Assignment
				if asg.JobID == "" {
					time.Sleep(50 * time.Millisecond)
					continue
				}

				// Small artificial delay to race with cancel / lease expiry
				time.Sleep(time.Duration(idxHash(asg.JobID)%2000) * time.Millisecond)

				// Report result
				resBody, _ := json.Marshal(map[string]any{
					"worker_id":   workerID,
					"lease_token": asg.LeaseToken,
					"exit_code":   0,
					"output":      "ok",
				})
				rResp, err := client.Post(fmt.Sprintf("http://%s/v1/jobs/%s/attempts/%s/result", addr, asg.JobID, asg.AttemptID), "application/json", bytes.NewReader(resBody))
				if err == nil {
					if rResp.StatusCode == http.StatusOK {
						commitsSuccess.Add(1)
					} else {
						commitsLost.Add(1)
					}
					rResp.Body.Close()
				}
			}
		}(wid)
	}

	// Concurrent cancellation goroutines targeting the 200 jobs
	var cancelWg sync.WaitGroup
	for _, jid := range jobIDs {
		if jid == "" {
			continue
		}
		cancelWg.Add(1)
		go func(targetID string) {
			defer cancelWg.Done()
			// Random delay up to 1.5s
			time.Sleep(time.Duration(idxHash(targetID)%1500) * time.Millisecond)
			resp, err := client.Post(fmt.Sprintf("http://%s/v1/jobs/%s/cancel", addr, targetID), "application/json", nil)
			if err == nil {
				if resp.StatusCode == http.StatusOK {
					cancelsSuccess.Add(1)
				} else {
					cancelsFailed.Add(1)
				}
				resp.Body.Close()
			}
		}(jid)
	}

	cancelWg.Wait()
	// Wait a bit for remaining jobs to settle
	time.Sleep(6 * time.Second)
	cancel()
	workerWg.Wait()

	fmt.Println("=== Race Phase Completed ===")
	fmt.Printf("Committed successes: %d, Committed lost/conflicts: %d\n", commitsSuccess.Load(), commitsLost.Load())
	fmt.Printf("Cancel successes: %d, Cancel conflicts/failed: %d\n", cancelsSuccess.Load(), cancelsFailed.Load())

	// Audit all 200 jobs in database
	stateCounts := make(map[string]int)
	for _, jid := range jobIDs {
		if jid == "" {
			continue
		}
		resp, err := client.Get(fmt.Sprintf("http://%s/v1/jobs/%s", addr, jid))
		if err != nil {
			fmt.Printf("GetJob %s error: %v\n", jid, err)
			continue
		}
		var j struct {
			State string `json:"state"`
		}
		_ = json.NewDecoder(resp.Body).Decode(&j)
		resp.Body.Close()
		stateCounts[j.State]++
	}

	fmt.Println("=== Final Job States across 200 jobs ===")
	for st, count := range stateCounts {
		fmt.Printf("  %s: %d\n", st, count)
	}
}

func idxHash(s string) int {
	h := 0
	for _, c := range s {
		h = (h*31 + int(c)) & 0x7fffffff
	}
	return h
}
