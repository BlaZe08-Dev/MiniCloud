#!/usr/bin/env bash
set -euo pipefail

URL=$(grep -E '^MINICLOUD_TEST_DATABASE_URL=' .env | cut -d= -f2-)
PORT=18080
ADDR="127.0.0.1:${PORT}"

# Start coordinator in background
MINICLOUD_STORAGE_DRIVER=postgres \
MINICLOUD_DATABASE_URL="$URL" \
MINICLOUD_HTTP_ADDR="$ADDR" \
bin/coordinator > /tmp/coord-a.log 2>&1 &
PID=$!
trap "kill -9 $PID >/dev/null 2>&1 || true" EXIT

# Wait for readiness
for i in {1..20}; do
  if curl -s "http://${ADDR}/readyz" >/dev/null 2>&1; then
    break
  fi
  sleep 0.2
done

echo "=== Attack A1: JSON string with \u0000 escape ==="
curl -i -s -X POST "http://${ADDR}/v1/jobs" \
  -H "Content-Type: application/json" \
  -d '{"task_type":"echo","payload":{"message":"test\u0000null"}}'
echo ""

echo "=== Attack A2: Invalid UTF-8 in payload ==="
python3 -c '
import urllib.request
req = urllib.request.Request("http://127.0.0.1:18080/v1/jobs", data=b"{\"task_type\":\"echo\",\"payload\":{\"bad\":\"\xff\xfe\"}}", headers={"Content-Type": "application/json"})
try:
    with urllib.request.urlopen(req) as resp:
        print("Status:", resp.status)
        print("Body:", resp.read().decode("utf-8", errors="replace"))
except urllib.error.HTTPError as e:
    print("Status:", e.code)
    print("Body:", e.read().decode("utf-8", errors="replace"))
'
echo ""

echo "=== Attack A3: Lone surrogate escape (\uD800) in payload ==="
curl -i -s -X POST "http://${ADDR}/v1/jobs" \
  -H "Content-Type: application/json" \
  -d '{"task_type":"echo","payload":{"msg":"\uD800"}}'
echo ""

echo "=== Attack A4: Duplicate JSON keys in payload ==="
curl -i -s -X POST "http://${ADDR}/v1/jobs" \
  -H "Content-Type: application/json" \
  -d '{"task_type":"echo","payload":{"message":"first","message":"second"}}'
echo ""

echo "=== Attack A5: Payload at 64 KiB field limit boundary vs 1 MiB HTTP limit ==="
# Field limit for payload is 64 KiB (docs/security/field-limits.md)
# HTTP body limit is 1 MiB (1048576 bytes)
python3 -c '
import urllib.request, json
# Exactly 64 KiB
pad64 = "a" * (65536 - 30)
data64 = json.dumps({"task_type": "echo", "payload": {"message": pad64}}).encode()
req = urllib.request.Request("http://127.0.0.1:18080/v1/jobs", data=data64, headers={"Content-Type": "application/json"})
try:
    with urllib.request.urlopen(req) as resp:
        print("64 KiB status:", resp.status)
        print("64 KiB body:", resp.read().decode())
except urllib.error.HTTPError as e:
    print("64 KiB error:", e.code, e.read().decode())

# Over 64 KiB payload (e.g. 70 KiB)
pad70 = "a" * 70000
data70 = json.dumps({"task_type": "echo", "payload": {"message": pad70}}).encode()
req = urllib.request.Request("http://127.0.0.1:18080/v1/jobs", data=data70, headers={"Content-Type": "application/json"})
try:
    with urllib.request.urlopen(req) as resp:
        print("70 KiB status:", resp.status)
except urllib.error.HTTPError as e:
    print("70 KiB error:", e.code, e.read().decode())

# Exactly 1 MiB HTTP body
pad1M = "a" * (1048576 - 50)
data1M = json.dumps({"task_type": "echo", "payload": {"message": "ok"}, "pad": pad1M}).encode()
req = urllib.request.Request("http://127.0.0.1:18080/v1/jobs", data=data1M[:1048576], headers={"Content-Type": "application/json"})
try:
    with urllib.request.urlopen(req) as resp:
        print("1 MiB HTTP body status:", resp.status)
except urllib.error.HTTPError as e:
    print("1 MiB HTTP body error:", e.code, e.read().decode())

# 1 MiB + 1 byte
data1M_over = data1M[:1048576] + b"x"
req = urllib.request.Request("http://127.0.0.1:18080/v1/jobs", data=data1M_over, headers={"Content-Type": "application/json"})
try:
    with urllib.request.urlopen(req) as resp:
        print("1 MiB + 1 byte status:", resp.status)
except urllib.error.HTTPError as e:
    print("1 MiB + 1 byte error:", e.code, e.read().decode())
'
echo ""
