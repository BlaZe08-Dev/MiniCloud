// NormalizeWorkerErrorClass validates the error class a worker reported with an outcome and
// returns the class to store. A worker may report only PERMANENT, TRANSIENT or UNKNOWN, and only
// with a failure; an omitted class on a failure defaults to UNKNOWN. LEASE_EXPIRED is assigned
// by the coordinator's lease sweeper alone and is rejected here, as is any other value or a class
// on a success (protocol section 6, ADR-0015). The rejected value is never echoed back.
func NormalizeWorkerErrorClass(c ErrorClass, failed bool) (ErrorClass, error) {
	if !failed {
		if c != "" {
			return "", fmt.Errorf("%w: error_classification is only allowed on a failed outcome", ErrInvalidRequest)
		}
		return "", nil
	}
	switch c {
	case "":
		return ClassUnknown, nil
	case ClassPermanent, ClassTransient, ClassUnknown:
		return c, nil
	default:
		return "", fmt.Errorf("%w: error_classification must be one of PERMANENT, TRANSIENT, UNKNOWN", ErrInvalidRequest)
	}
}

// Classify maps the error of a failed Execute call to the failure class a worker reports with its
// result (ADR-0015, docs/protocol.md section 6):
//
//   - PERMANENT: validation-style failures of an allowlisted task (invalid or oversized payload,
//     unknown task type). The same payload fails the same way, so the job must not be retried.
//   - TRANSIENT: the task hit its execution deadline.
//   - UNKNOWN: everything else, including a cooperative cancellation; retryable by default.
//
// Success (a nil error) has no class. LEASE_EXPIRED is the coordinator's alone and is never
// produced here.
func Classify(err error) domain.ErrorClass {
	switch {
	case err == nil:
		return ""
	case errors.Is(err, domain.ErrInvalidRequest),
		errors.Is(err, domain.ErrUnknownTaskType),
		errors.Is(err, domain.ErrPayloadTooLarge):
		return domain.ClassPermanent
	case errors.Is(err, domain.ErrTaskTimeout), errors.Is(err, context.DeadlineExceeded):
		return domain.ClassTransient
	default:
		return domain.ClassUnknown
	}
}

// ErrServiceUnavailable indicates an upstream coordinator or database outage (HTTP 503/5xx).

// ComputeBackoffWithJitter calculates exponential backoff with full jitter.
func ComputeBackoffWithJitter(failureCount int, base, max time.Duration) time.Duration {
	if failureCount <= 0 {
		return base
	}
	multiplier := 1 << (failureCount - 1)
	if multiplier > 16 {
		multiplier = 16
	}
	dur := base * time.Duration(multiplier)
	if dur > max {
		dur = max
	}
	jitterFrac := 0.5 + rand.Float64()*0.5
	return time.Duration(float64(dur) * jitterFrac)
}

	NewTimer(d time.Duration) Timer

	Stop() bool

func (a *Agent) renewLoop(ctx context.Context, run *attemptRun, asg *coordinator.WorkAssignment, interval time.Duration, cancelTask context.CancelFunc) {
	failures := 0
	wait := interval
	for {
		t := a.clock.NewTimer(wait)
		select {
		case <-ctx.Done():
			t.Stop()
			return
		case <-t.C():
		}
		switch a.renewOnce(ctx, asg, interval) {
		case renewOK:
			failures, wait = 0, interval
		case renewCancel:
			slog.Info("cancellation requested by coordinator; cancelling task", "job_id", asg.JobID, "attempt_id", asg.AttemptID)
			cancelTask()
			return
		case renewLost:
			slog.Warn("coordinator declared the attempt lost; aborting task and suppressing the result", "job_id", asg.JobID, "attempt_id", asg.AttemptID)
			run.markLost()
			cancelTask()
			return
		case renewDisable:
			return
		case renewTransient:
			if ctx.Err() != nil {
				return
			}
			failures++
			wait = min(ComputeBackoffWithJitter(failures, renewBackoffBase, interval), interval)
		}
	}
}

== ADR-0016 Race 2 + cancel renew behaviour
128:#### Race 2: Cancellation vs Lease Expiry (Zombie / Unresponsive Worker)
129-- **Scenario**: Job is in `CANCEL_REQUESTED`. Worker hangs in non-cooperative task (ignoring context cancellation), loses network connectivity, or dies before reporting outcome.
130-- **Resolution**:
131-  - The renewal response on `CANCEL_REQUESTED` did not extend the lease, and the worker ceased renewing. Therefore, the lease expires within at most one lease duration (30 seconds) from the last renewal.
132-  - The coordinator lease sweeper (ADR-0014) locks `jobs -> workers -> attempts`.
133-  - Seeing `job.state == 'CANCEL_REQUESTED'` and `lease_expires_at < clock_timestamp()`:
134-    1. Sweeper transitions attempt to `CANCELLED`.
135-    2. Sweeper transitions job: `CANCEL_REQUESTED` -> `CANCELLED` (terminal).
136-    3. Sweeper **suppresses retries**: even if `attempt_no < max_attempts`, a cancelled job is never retried (never `RETRYING`).
137-    4. Event: `JobEventCancelled` with detail `{"resolved_by": "lease_sweeper", "reason": "worker unresponsive"}`.
138-  - **Sweeper Rule for Already-Terminal Jobs**:
139-    If the sweeper encounters an attempt whose job is already in a terminal state (`CANCELLED`, `SUCCEEDED`, or `FAILED`), it closes the attempt: to `CANCELLED` if the job is `CANCELLED`, or to `LOST` otherwise. The job state remains unchanged, the worker slot is released, and a diagnostic event is recorded.
