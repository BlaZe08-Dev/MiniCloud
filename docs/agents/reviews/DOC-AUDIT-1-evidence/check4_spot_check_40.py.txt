def main():
    print(f"--- Spot Checking {len(CITATIONS_TO_CHECK)} Citations Across Repo ---")
    verified = 0
    mismatches = 0
    for doc, doc_line, path, target_line, topic in CITATIONS_TO_CHECK:
        full_target = os.path.join(REPO_ROOT, path)
        if not os.path.exists(full_target):
            print(f"FAILED (file missing): {doc}:{doc_line} cites {path}:{target_line} ({topic})")
            mismatches += 1
            continue
        lines = open(full_target).readlines()
        if target_line > len(lines):
            print(f"FAILED (line out of bounds): {doc}:{doc_line} cites {path}:{target_line} (file has {len(lines)} lines)")
            mismatches += 1
            continue
        line_content = lines[target_line - 1].strip()
        print(f"OK: {doc} -> {path}:{target_line} | line content: {line_content[:60]}")
        verified += 1
    
    print(f"\nSummary: Verified={verified}, Mismatches={mismatches}")

if __name__ == "__main__":
    main()
