#!/usr/bin/env bash
# V2-0b audit, item 5: repeat the two flaky worker tests on the branch tip (b5aec6f, ~/MiniCloud-claude)
# and on main 26b440a (~/mc-main-claude, separate worktree), one run at a time, nothing else testing.
# No database: MINICLOUD_TEST_DATABASE_URL is unset (both tests are fake-clock / real-clock worker tests).
# usage: flaky.sh LOGDIR
set -u
LOG=$1; mkdir -p "$LOG"
SUM="$LOG/flaky-summary.txt"; : > "$SUM"
run() { # label tree flags... (last arg is the -run regex)
  local label=$1 tree=$2; shift 2
  local f="$LOG/$label.log"
  local t0=$(date +%s)
  (cd "$tree" && env -u MINICLOUD_TEST_DATABASE_URL go test -v "$@" ./internal/worker/ > "$f" 2>&1); local rc=$?
  local t1=$(date +%s)
  printf '%-34s tree=%s rev=%s rc=%s PASS=%s FAIL=%s races=%s secs=%s load_before=%s\n' "$label" "$tree" \
    "$(git -C "$tree" rev-parse --short HEAD)" "$rc" "$(grep -c '^--- PASS' "$f")" "$(grep -c '^--- FAIL' "$f")" \
    "$(grep -c 'WARNING: DATA RACE' "$f")" "$((t1 - t0))" "$LOAD" >> "$SUM"
  grep -E -A3 '^--- FAIL' "$f" | head -20 >> "$SUM"
}
B=$HOME/MiniCloud-claude; M=$HOME/mc-main-claude
R='^TestConcurrency_RenewalPanicTreatsLeaseAsLost$'; P='^TestPollTiming_RealClockSleeps$'
for tree in B M; do
  T=${!tree}
  LOAD=$(cut -d' ' -f1-3 /proc/loadavg | tr ' ' '/'); run "renewal-count200-$tree" "$T" -count=200 -run "$R"
  LOAD=$(cut -d' ' -f1-3 /proc/loadavg | tr ' ' '/'); run "renewal-race-count100-$tree" "$T" -race -count=100 -run "$R"
  LOAD=$(cut -d' ' -f1-3 /proc/loadavg | tr ' ' '/'); run "polltiming-count50-$tree" "$T" -count=50 -run "$P"
  LOAD=$(cut -d' ' -f1-3 /proc/loadavg | tr ' ' '/'); run "polltiming-race-count50-$tree" "$T" -race -count=50 -run "$P"
done
cat "$SUM"
