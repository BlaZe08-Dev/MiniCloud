import re,subprocess,sys,difflib
repo=sys.argv[1]
files=["CHANGELOG.md","docs/PROJECT_STATE.md","docs/security/findings.md","docs/security/test-plan.md","docs/security/threat-model.md"]
def defs(name):
    r=subprocess.run(["git","-C",repo,"grep","-n","-E","^func "+name+r"\(","--","*.go"],capture_output=True,text=True).stdout.split("\n")
    return sorted(set(x.split(":")[0] for x in r if x))
P=re.compile(r'`(Test\w+)(?:/[\w]+)?`[^`\n]{0,40}?`((?:internal|tests|cmd)/[\w/.\-]+\.go)(?::[\d-]+)?`')
T=re.compile(r'`(Test\w+)')
PA=re.compile(r'`((?:internal|tests|cmd)/[\w/.\-]+_test\.go)')
cache={}
def D(n):
    if n not in cache:cache[n]=defs(n)
    return cache[n]
bad=0;pairs=0;unpaired=[]
for f in files:
    old=subprocess.run(["git","-C",repo,"show","origin/main:"+f],capture_output=True,text=True).stdout.split("\n")
    new=open(repo+"/"+f).read().split("\n")
    sm=difflib.SequenceMatcher(None,old,new,autojunk=False)
    for tag,i1,i2,j1,j2 in sm.get_opcodes():
        if tag=="equal":continue
        for k in range(j1,j2):
            line=new[k]
            ps=P.findall(line)
            for n,p in ps:
                pairs+=1
                d=D(n)
                if p not in d:
                    bad+=1;print("BAD",f,k+1,n,"cites",p,"defined in",d)
            paired=set(n for n,_ in ps)
            for n in T.findall(line):
                if n not in paired and n not in unpaired: pass
            # unpaired names / paths
            un=[n for n in set(T.findall(line)) if n not in paired]
            if un: unpaired.append((f,k+1,un,sorted(set(PA.findall(line)))))
print("pairs",pairs,"bad",bad)
for u in unpaired:
    f,l,names,paths=u
    print("UNPAIRED",f,l)
    for n in names: print("   ",n,"defs:",D(n),"| paths on line:",paths)
