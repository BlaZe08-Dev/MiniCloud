func (h *Handler) HandleSubmitJob(w http.ResponseWriter, r *http.Request) {
	var req SubmitJobRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeError(w, http.StatusBadRequest, "MALFORMED_JSON", "malformed JSON request body")
		return
	}

	if req.TaskType == "" {
		writeError(w, http.StatusBadRequest, "INVALID_REQUEST", "task_type is required")
		return
	}

	maxAttempts := coordinator.DefaultMaxAttempts
	if req.MaxAttempts != nil {
		if *req.MaxAttempts < coordinator.MinMaxAttempts || *req.MaxAttempts > coordinator.MaxMaxAttempts {
			writeError(w, http.StatusBadRequest, "INVALID_REQUEST", fmt.Sprintf("max_attempts must be between %d and %d", coordinator.MinMaxAttempts, coordinator.MaxMaxAttempts))
			return
		}
		maxAttempts = *req.MaxAttempts
	}

	timeoutSec := coordinator.DefaultTimeoutSeconds
	if req.TimeoutSeconds != nil {
		if *req.TimeoutSeconds < coordinator.MinTimeoutSeconds || *req.TimeoutSeconds > coordinator.MaxTimeoutSeconds {
			writeError(w, http.StatusBadRequest, "INVALID_REQUEST", fmt.Sprintf("timeout_seconds must be between %d and %d", coordinator.MinTimeoutSeconds, coordinator.MaxTimeoutSeconds))
			return
		}
		timeoutSec = *req.TimeoutSeconds
	}

	job, err := h.coord.SubmitJob(r.Context(), req.TaskType, req.Payload, timeoutSec, maxAttempts)
	if err != nil {
		status, code, msg := mapDomainErrorToHTTP(err)
		writeError(w, status, code, msg)
		return
	}

	writeJSON(w, http.StatusCreated, job)
}

}

// ComputeBackoffWithJitter calculates exponential backoff with full jitter.
func ComputeBackoffWithJitter(failureCount int, base, max time.Duration) time.Duration {
	if failureCount <= 0 {
		return base
	}
	multiplier := 1 << (failureCount - 1)
	if multiplier > 16 {
		multiplier = 16
	}
	dur := base * time.Duration(multiplier)
	if dur > max {
		dur = max
	}
	jitterFrac := 0.5 + rand.Float64()*0.5
	return time.Duration(float64(dur) * jitterFrac)
}

func handleSubmit(args []string) {
	fs := flag.NewFlagSet("submit", flag.ExitOnError)
	taskType := fs.String("task", "", "Allowlisted task type (echo, sleep, sha256_loop)")
	payload := fs.String("payload", "{}", "JSON payload string")
	timeoutSec := fs.Int("timeout", 60, "Task execution timeout in seconds")
	maxAttempts := fs.Int("max-attempts", 3, "Maximum execution attempts")
	coordURL := fs.String("coordinator-url", getCoordinatorURL(), "Coordinator URL")

	_ = fs.Parse(args)

	if *taskType == "" {
		fmt.Fprintln(os.Stderr, "Error: --task is required")
		fs.Usage()
		os.Exit(1)
	}

	if !json.Valid([]byte(*payload)) {
		fmt.Fprintln(os.Stderr, "Error: --payload must be valid JSON")
		os.Exit(1)
	}

	reqBody, _ := json.Marshal(map[string]any{
		"task_type":       *taskType,
		"payload":         json.RawMessage(*payload),
		"timeout_seconds": *timeoutSec,
		"max_attempts":    *maxAttempts,
	})

	resp, err := http.Post(*coordURL+"/v1/jobs", "application/json", bytes.NewReader(reqBody))
	if err != nil {
		fmt.Fprintf(os.Stderr, "Error submitting job: %v\n", err)
		os.Exit(1)
	}
	defer resp.Body.Close()
