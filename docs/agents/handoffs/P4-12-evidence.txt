set -e; \
if [ -n "${MINICLOUD_TEST_DATABASE_URL+x}" ]; then \
	URL="$MINICLOUD_TEST_DATABASE_URL"; \
elif [ -n "postgres://blazeee:minicloud@127.0.0.1:5433/minicloud_test?sslmode=disable" ]; then \
	URL="postgres://blazeee:minicloud@127.0.0.1:5433/minicloud_test?sslmode=disable"; \

	URL=$(grep -E '^MINICLOUD_TEST_DATABASE_URL=' .env | cut -d= -f2-); \

	exit 1; \
fi; \
MINICLOUD_TEST_DATABASE_URL="$URL" go test -v -race -count=1 -timeout 20m -run '^(TestFault_|TestAbuse_)' ./tests/faultinject/ ./tests/abuse/
