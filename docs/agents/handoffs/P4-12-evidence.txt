==================== make test-security own name (make -n) ====================
$ MINICLOUD_TEST_DATABASE_URL='postgres://127.0.0.1:5433/minicloud_gemini_test' make -n test-security
exit code: 0
stdout:
if [ ! -f tests/security-tests.list ]; then echo "tests/security-tests.list missing" >&2; exit 1; fi
set -e; \
if [ -n "${MINICLOUD_TEST_DATABASE_URL+x}" ]; then \
	URL="$MINICLOUD_TEST_DATABASE_URL"; \
elif [ -n "postgres://127.0.0.1:5433/minicloud_gemini_test" ]; then \
	URL="postgres://127.0.0.1:5433/minicloud_gemini_test"; \
elif [ -f .env ]; then \
	URL=$(grep -E '^MINICLOUD_TEST_DATABASE_URL=' .env | cut -d= -f2-); \
else \
	URL=""; \
fi; \
ALLOW_SHARED_DB="${ALLOW_SHARED_DB:-}" tests/require-own-db.sh "$URL"; \
ACT=$(mktemp); trap 'rm -f "$ACT"' EXIT; \
env -u MINICLOUD_TEST_DATABASE_URL go test -list '^(TestFault_|TestAbuse_)' ./tests/faultinject/ ./tests/abuse/ | grep '^Test' | LC_ALL=C sort > "$ACT" || true; \
if ! diff -q tests/security-tests.list "$ACT" >/dev/null; then \
	echo "FALSE-GREEN GUARD: pattern '^(TestFault_|TestAbuse_)' does not match tests/security-tests.list" >&2; \
	echo "missing (in the list, not found by go test -list):" >&2; LC_ALL=C comm -23 tests/security-tests.list "$ACT" | sed 's/^/  /' >&2; \
	echo "extra (found by go test -list, not in the list):" >&2; LC_ALL=C comm -13 tests/security-tests.list "$ACT" | sed 's/^/  /' >&2; \
	exit 1; \
fi; \
MINICLOUD_TEST_DATABASE_URL="$URL" go test -v -race -count=1 -timeout 20m -run '^(TestFault_|TestAbuse_)' ./tests/faultinject/ ./tests/abuse/
