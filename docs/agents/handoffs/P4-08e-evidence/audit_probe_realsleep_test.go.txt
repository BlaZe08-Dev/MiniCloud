// Audit probe (P4-08e review, item 2). Copied into internal/worker/ of a throwaway worktree as
// audit_probe_realsleep_test.go and run alone. It uses only API that exists on origin/main and on
// the P4-08e branch, and never calls SetClock, so Agent.Run sleeps on the production clock.
// The fake coordinator records the arrival time of each poll; the gap between two polls is the
// real sleep plus one local round trip. Sequence: 5 x 204, 4 x 500, 1 x 503 WORKER_UNHEALTHY,
// then 204 (cancel). Bounds: after a 204 or WORKER_UNHEALTHY the gap is in [200 ms, 200 ms + eps];
// after the k-th 500 it is in [0.5 x c_k, c_k + eps], c_k = min(200 ms x 2^(k-1), 3 s).
package worker_test

import (
	"context"
	"encoding/json"
	"io"
	"log/slog"
	"net/http"
	"net/http/httptest"
	"sync"
	"testing"
	"time"

	"github.com/BlaZe08-Dev/Minicloud/internal/config"
	"github.com/BlaZe08-Dev/Minicloud/internal/worker"
)

func TestAuditProbe_RealClockPollSleeps(t *testing.T) {
	prev := slog.Default()
	slog.SetDefault(slog.New(slog.NewTextHandler(io.Discard, nil)))
	t.Cleanup(func() { slog.SetDefault(prev) })

	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	var mu sync.Mutex
	var at []time.Time
	ts := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		switch r.URL.Path {
		case "/v1/workers/register", "/v1/workers/w-probe/heartbeat":
			_ = json.NewEncoder(w).Encode(map[string]string{"status": "ACK"})
		case "/v1/workers/w-probe/poll":
			mu.Lock()
			at = append(at, time.Now())
			n := len(at)
			mu.Unlock()
			switch {
			case n <= 5:
				w.WriteHeader(http.StatusNoContent)
			case n <= 9:
				w.WriteHeader(http.StatusInternalServerError)
			case n == 10:
				w.WriteHeader(http.StatusServiceUnavailable)
				_ = json.NewEncoder(w).Encode(map[string]any{"error": map[string]any{"code": "WORKER_UNHEALTHY", "message": "x"}})
			default:
				cancel()
				w.WriteHeader(http.StatusNoContent)
			}
		default:
			http.NotFound(w, r)
		}
	}))
	defer ts.Close()

	agent := worker.NewAgent(&config.WorkerConfig{WorkerID: "w-probe", CoordinatorURL: ts.URL, HeartbeatInterval: time.Hour, TotalSlots: 1}, nil)
	done := make(chan error, 1)
	go func() { done <- agent.Run(ctx) }()
	select {
	case <-done:
	case <-time.After(30 * time.Second):
		t.Fatalf("Run did not finish")
	}

	mu.Lock()
	defer mu.Unlock()
	if len(at) < 11 {
		t.Fatalf("only %d polls", len(at))
	}
	const eps = 150 * time.Millisecond
	idle := 200 * time.Millisecond
	for i := 1; i <= 10; i++ {
		gap := at[i].Sub(at[i-1])
		lo, hi, kind := idle, idle+eps, "idle"
		if i >= 6 && i <= 9 { // gap after the (i-5)-th 500
			c := 200 * time.Millisecond << (i - 6)
			if c > 3*time.Second {
				c = 3 * time.Second
			}
			lo, hi, kind = c/2, c+eps, "backoff"
		}
		if i == 10 {
			kind = "unhealthy"
		}
		t.Logf("gap %2d (%-9s) %8.1f ms  bound [%v, %v]", i, kind, float64(gap.Microseconds())/1000, lo, hi)
		if gap < lo || gap > hi {
			t.Errorf("gap %d = %v outside [%v, %v]", i, gap, lo, hi)
		}
	}
}
