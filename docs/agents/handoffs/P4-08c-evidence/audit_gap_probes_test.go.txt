// A01: the persisted cursor must move on every pass, not only on its first insert. Three passes
// for one type over w_a, w_b, w_c, a restart before the third: the third pick is w_c, and the
// stored cursor row follows each pick.
func TestAudit_Cursor_UpdatePathPersistsAcrossPasses(t *testing.T) {
	ctx := context.Background()
	h, pool, repo := newPostgresAssignmentHarnessWithRepo(t)
	for _, id := range []string{"w_a", "w_b", "w_c"} {
		storagetest.AssignAddWorker(t, h, id, 1, "echo")
		storagetest.AssignPoll(t, h, id)
	}
	cursorRow := func() string {
		var w *string
		if err := pool.QueryRow(ctx, `SELECT last_worker_id FROM assignment_cursor WHERE task_type = 'echo'`).Scan(&w); err != nil || w == nil {
			t.Fatalf("cursor row: %v", err)
		}
		return *w
	}
	for i, want := range []string{"w_a", "w_b", "w_c"} {
		if i == 2 {
			h.Repo = postgres.NewRepository(pool) // restart before the third pass
		}
		id := []string{"job_1", "job_2", "job_3"}[i]
		storagetest.AssignSubmit(t, h, id, "echo")
		storagetest.AssignRunPass(t, h)
		if got := reservedTo(t, pool, id); got != want {
			t.Fatalf("pass %d reserved %s to %q, want %q", i+1, id, got, want)
		}
		if c := cursorRow(); c != want {
			t.Fatalf("pass %d left the stored cursor at %q, want %q", i+1, c, want)
		}
		if _, err := repo.CancelJob(ctx, id); err != nil {
			t.Fatalf("CancelJob(%s): %v", id, err)
		}
	}
}

// A01b: within one pass each decision starts from the previous decision's pick; the JOB_ASSIGNED
// payload's cursor_before records it (ADR-0017 Section 2.1 event shape).
func TestAudit_Event_CursorBeforeAdvancesWithinPass(t *testing.T) {
	ctx := context.Background()
	h, _, repo := newPostgresAssignmentHarnessWithRepo(t)
	for _, id := range []string{"w_a", "w_b"} {
		storagetest.AssignAddWorker(t, h, id, 1, "echo")
		storagetest.AssignPoll(t, h, id)
	}
	storagetest.AssignSubmit(t, h, "job_1", "echo")
	storagetest.AssignSubmit(t, h, "job_2", "echo")
	if res := storagetest.AssignRunPass(t, h); res.Reserved != 2 {
		t.Fatalf("control: reserved %d, want 2", res.Reserved)
	}
	evs, err := repo.ListEventsByJob(ctx, "job_2")
	if err != nil {
		t.Fatal(err)
	}
	for _, e := range evs {
		if e.EventType == domain.JobEventAssigned {
			if cb, _ := e.Details["cursor_before"].(string); cb != "w_a" {
				t.Fatalf("job_2 JOB_ASSIGNED cursor_before=%q selected=%v, want cursor_before w_a", cb, e.Details["selected"])
			}
			return
		}
	}
	t.Fatalf("no JOB_ASSIGNED event for job_2")
}

// A05: the pass reserves the oldest dispatchable job of a type first (ADR-0017 Section 2.5): with
// one eligible worker and two jobs of one type, the older job gets the reservation.
func TestAudit_Pass_ReservesOldestOfTypeFirst(t *testing.T) {
	h, pool, _ := newPostgresAssignmentHarnessWithRepo(t)
	storagetest.AssignAddWorker(t, h, "w_a", 1, "echo")
	storagetest.AssignPoll(t, h, "w_a")
	storagetest.AssignSubmit(t, h, "job_1_old", "echo")
	storagetest.AssignSubmit(t, h, "job_2_new", "echo")
	storagetest.AssignRunPass(t, h)
	if o, n := reservedTo(t, pool, "job_1_old"), reservedTo(t, pool, "job_2_new"); o != "w_a" || n != "" {
		t.Fatalf("old->%q new->%q, want the older job reserved to w_a and the newer one unreserved", o, n)
	}
}

// A06b: a reservation whose window has passed is not live for the fallback arm (ADR-0017 Section
// 3.2 step 2): a job at the lapse limit with a reservation expired 1 s ago (not yet revoked) is
// claimable by another worker with reason lapses.
func TestAudit_Fallback_ExpiredUnrevokedReservationIsClaimable(t *testing.T) {
	ctx := context.Background()
	h, _, repo := newPostgresAssignmentHarnessWithRepo(t)
	storagetest.AssignAddWorker(t, h, "w_1", 1, "echo")
	storagetest.AssignAddWorker(t, h, "w_2", 1, "echo")
	storagetest.AssignPoll(t, h, "w_2")
	storagetest.AssignSubmit(t, h, "job_1", "echo")
	h.SetLapses(t, "job_1", storagetest.AssignTestMaxLapses)
	h.SetReservation(t, "job_1", "w_1", -1*time.Second)
	claim, err := repo.ClaimForPoll(ctx, "w_2", storagetest.AssignTestClaimOpts(false))
	if err != nil {
		t.Fatalf("w_2 claim of a lapse-limit job whose reservation expired 1 s ago: %v", err)
	}
	if claim.Job.ID != "job_1" || claim.FallbackReason != domain.FallbackReasonLapses {
		t.Fatalf("claimed %s reason %q, want job_1 by lapses", claim.Job.ID, claim.FallbackReason)
	}
}
