import subprocess,sys,os
wt=os.path.expanduser('~/mc-mut-claude'); out=sys.argv[1]
M=[
("E01","cap removed (pending never ends)","internal/coordinator/sweeper.go","\t\tpendingUntil = s.pendingSince.Add(s.pendingCap)\n","\t\t_ = s.pendingCap\n"),
("E02","cap one sweep interval too long (+5s)","internal/coordinator/sweeper.go","pendingUntil = s.pendingSince.Add(s.pendingCap)","pendingUntil = s.pendingSince.Add(s.pendingCap + 5*time.Second)"),
("E03","cap one sweep interval too short (-5s)","internal/coordinator/sweeper.go","pendingUntil = s.pendingSince.Add(s.pendingCap)","pendingUntil = s.pendingSince.Add(s.pendingCap - 5*time.Second)"),
("E04","failed sweep no longer marks grace pending (A05)","internal/coordinator/sweeper.go","\t\tw.state.failedSince = true\n\t\tw.svc.publishPassGrace(domain.SweepGraceState{}, false, w.window)\n\t\treturn err\n\t}\n\t// The assignment","\t\tw.state.failedSince = true\n\t\treturn err\n\t}\n\t// The assignment"),
("E05","trigger buffer 64 (A04)","internal/coordinator/service.go","make(chan struct{}, 1)","make(chan struct{}, 64)"),
("E06","ERROR window 50 s (A03)","internal/coordinator/assignment_loop.go","const StaleHeartbeatLogEvery = time.Minute","const StaleHeartbeatLogEvery = 50 * time.Second"),
("E07","real yield does not wait (A02)","internal/coordinator/assignment_loop.go","time.NewTimer(d)","time.NewTimer(0 * d)"),
("E08","LapseCounts cap boundary inclusive","internal/storage/assignment.go","now.Before(p.GracePendingUntil)","!now.After(p.GracePendingUntil)"),
("E09","pending cap by cause swapped (startup uses outage window)","internal/coordinator/sweeper.go","s.publishPassGrace(domain.SweepGraceState{}, false, s.startupGrace)","s.publishPassGrace(domain.SweepGraceState{}, false, s.outageWindow)"),
]
only=sys.argv[2:]
def sh(c): return subprocess.run(c,shell=True,cwd=wt,capture_output=True,text=True)
env="env -u MINICLOUD_TEST_DATABASE_URL "
for mid,name,f,a,b in M:
    if only and mid not in only: continue
    p=os.path.join(wt,f); s=open(p).read()
    if s.count(a)!=1: print(mid,"PATTERN COUNT",s.count(a)); continue
    open(p,'w').write(s.replace(a,b))
    d=sh("git diff"); open(f"{out}/mut-{mid}.diff","w").write(d.stdout)
    r=sh(env+"go test -count=1 ./internal/coordinator ./internal/storage 2>&1")
    open(f"{out}/mut-{mid}-red.log","w").write(r.stdout)
    fails=[l for l in r.stdout.splitlines() if l.startswith("--- FAIL")]
    sh("git checkout -- .")
    g=sh(env+"go test -count=1 ./internal/coordinator ./internal/storage 2>&1")
    open(f"{out}/mut-{mid}-green.log","w").write(g.stdout)
    print(mid,name,"| red:", "KILLED" if fails else ("COMPILE/OTHER" if "FAIL" in r.stdout else "SURVIVED"), fails[:3], "| green rc", g.returncode)
