import re, subprocess, difflib

repo = "."
files = ["CHANGELOG.md", "docs/PROJECT_STATE.md", "docs/security/findings.md", "docs/security/test-plan.md", "docs/security/threat-model.md"]
C = re.compile(r"(?:internal|tests|cmd)/[\w/.\-]+_test\.go:\d+(?:-\d+)?")
removed = []
kept = []
odd = []

for f in files:
    old = subprocess.run(["git", "show", "fc451c9:" + f], capture_output=True, text=True).stdout.split("\n")
    new = open(f).read().split("\n")
    sm = difflib.SequenceMatcher(None, old, new, autojunk=False)
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag == "equal": continue
        if not (tag == "replace" and i2 - i1 == j2 - j1):
            odd.append((f, tag, i1 + 1))
            continue
        for k, (a, b) in enumerate(zip(old[i1:i2], new[j1:j2])):
            parts = C.split(a)
            cs = C.findall(a)
            if not cs:
                odd.append((f, "nocite", i1 + k + 1, a[:100]))
                continue
            pat = "^" + "".join(re.escape(p) + ("(" + re.escape(c) + "|" + re.escape(c.split(":")[0]) + ")" if n < len(cs) else "") for n, (p, c) in enumerate(zip(parts, cs + [None]))) + "$"
            m = re.match(pat, b)
            if not m:
                odd.append((f, "notsubset", i1 + k + 1, a[:80]))
                continue
            for c, g in zip(cs, m.groups()):
                (removed if ":" not in g else kept).append((f, i1 + k + 1, c))

from collections import Counter
counts = Counter(r[0] for r in removed)
print("REMOVED total:", len(removed))
for f in files:
    print(f"  {f}: {counts[f]}")
